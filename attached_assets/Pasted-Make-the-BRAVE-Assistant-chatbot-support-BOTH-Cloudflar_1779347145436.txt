  async function callCloudflareAI(
    messages: ChatMsg[],
    apiToken: string,
    accountId: string,
    log: Request["log"],
  ): Promise<string | null> {
    // 25s timeout (same shape as callCerebras). On non-OK or transport
    // failure return null; the route's retry-once flow will handle it.
    // Cloudflare's OpenAI-compatible response uses the same
    // choices[0].message.content shape Cerebras returns.
    POST `${CLOUDFLARE_BASE_URL}/${accountId}/ai/v1/chat/completions`
      Headers: Authorization: Bearer <token>, Content-Type: application/json
      Body: { model: CLOUDFLARE_MODEL, messages, temperature: 0.3,
              max_tokens: 700, response_format: { type: "json_object" } }
  }

3.3) Add a tiny in-memory cache + DB lookup for the active provider so
we don't hit the DB on every chatbot request:

  // Cached for 30 seconds — admin's "Save" applies on the next request
  // after the TTL expires, which is fast enough for a config switch.
  let cachedProvider: "cloudflare" | "cerebras" | null = null;
  let cachedProviderAt = 0;
  const PROVIDER_CACHE_TTL_MS = 30_000;

  async function getActiveProvider(): Promise<"cloudflare" | "cerebras"> {
    const now = Date.now();
    if (cachedProvider && now - cachedProviderAt < PROVIDER_CACHE_TTL_MS) {
      return cachedProvider;
    }
    try {
      const [row] = await db
        .select({ provider: programmeConfigTable.chatbotProvider })
        .from(programmeConfigTable)
        .limit(1);
      cachedProvider = row?.provider ?? "cloudflare";
    } catch {
      cachedProvider = "cloudflare";   // safe default on DB failure
    }
    cachedProviderAt = now;
    return cachedProvider;
  }

  // Exported so the admin PATCH handler can invalidate the cache
  // immediately when the provider is changed:
  export function invalidateChatbotProviderCache() {
    cachedProvider = null;
    cachedProviderAt = 0;
  }

(Add the necessary import of `db` and `programmeConfigTable` from

3.4) Replace the existing env-var check block in the POST "/chatbot/ask"
handler with provider-aware dispatch. Pseudocode of the new flow:

  const provider = await getActiveProvider();

  let upstreamCall: () => Promise<string | null>;
  if (provider === "cloudflare") {
    const apiToken  = process.env.CLOUDFLARE_API_TOKEN;
    const accountId = process.env.CLOUDFLARE_ACCOUNT_ID;
    if (!apiToken || !accountId) {
      req.log.warn("Cloudflare creds missing");
      res.status(200).json({
        answer:
          "The BRAVE assistant is not configured yet (missing Cloudflare credentials). Please ask an admin to add CLOUDFLARE_API_TOKEN and CLOUDFLARE_ACCOUNT_ID in Replit Secrets, or switch the provider on /admin/config.",
        suggestions: [],
      });
      return;
    }
    upstreamCall = () => callCloudflareAI(messages, apiToken, accountId, req.log);
  } else {
    const apiKey = process.env.CEREBRAS_API_KEY;
    if (!apiKey) {
      req.log.warn("Cerebras key missing");
      res.status(200).json({
        answer:
          "The BRAVE assistant is not configured yet (missing CEREBRAS_API_KEY). Please ask an admin to add it in Replit Secrets, or switch the provider on /admin/config.",
        suggestions: [],
      });
      return;
    }
    upstreamCall = () => callCerebras(messages, apiKey, req.log);
  }

  let content = await upstreamCall();
  let out = content ? parseModelReply(content) : null;

  if (!out) {
    // existing retry-once block — reuse the same `upstreamCall` closure
    // so the retry goes to the SAME provider (do not switch providers
    // mid-conversation).
    content = await upstreamCall();
    out = content ? parseModelReply(content) : null;
  }

  // ...rest of the handler (fallback message, default suggestions, res.json) is unchanged.

PART 4 — Admin API: GET/PATCH the active provider
==========================================================

Add two endpoints to the existing admin router
(artifacts/api-server/src/routes/admin.ts — or add to admin-notifications
router if cleaner; either way ADD-only, do not remove anything):

  GET  /api/admin/chatbot-provider
       → 200 { provider: "cloudflare" | "cerebras" }
       → admin-only (403 otherwise)
       → reads from programmeConfigTable.chatbotProvider, defaults to
         "cloudflare" if no row.

  PATCH /api/admin/chatbot-provider
       → body: { provider: "cloudflare" | "cerebras" }
       → admin-only (403 otherwise)
       → validates with z.enum(["cloudflare","cerebras"]); 400 on bad value.
       → updates programmeConfigTable.chatbotProvider on the single row
         (insert a default row if none exists).
       → calls invalidateChatbotProviderCache() so the new provider takes
         effect on the very next chatbot request.
       → writes auditLog "change_chatbot_provider" with the new value.
       → returns 200 { provider: "<new value>" }.

ACCEPTANCE
==========================================================

- A fresh DB has chatbotProvider="cloudflare" by default; the chatbot
  uses Cloudflare on first request.
- /admin/config shows the new "Chatbot LLM Provider" card at the bottom
  of the right column. Admin can pick a provider, click Save, and the
  next chat message goes to that provider (verified by the Replit logs:
  "Cloudflare request failed" vs "Cerebras request failed" only show
  the matching provider).
- Switching provider does NOT require a redeploy. Cache TTL is 30s; the
  PATCH endpoint also invalidates the cache so the switch is effectively
  immediate.
- If the selected provider's credentials are missing in env, the
  chatbot returns a clear "not configured" message naming which secrets
  to add — the chatbot does NOT silently fall back to the other
  provider.
- Audit log records "change_chatbot_provider" with old/new values when
  admin saves.
- Coordinators and students get 403 on both admin endpoints.
- `pnpm run typecheck` is clean.
- All existing routes still mount (every router.use() in
  routes/index.ts preserved — verify by hitting /api/campuses and
  /api/admin/journals after deploy; both must return 200, not 404).
